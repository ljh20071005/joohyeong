import re, json from pathlib import Path from collections import defaultdict import pandas as pd # === 1) Load & parse EXACTLY per user's rules === df = pd.read_excel("/mnt/data/timeta.xlsm", sheet_name="학생", engine="openpyxl") student_header_re = re.compile(r"^\s*(\d{5})\s+(.+?)\s*$") remove_time_re = re.compile(r"\(\s*\d+\s*시간\s*\)$") course_to_students = defaultdict(set) # "과목명 n반" -> {이름들} student_to_courses = defaultdict(set) # "이름" -> {"과목명 n반", ...} current_student = None for _, row in df.iterrows(): # A열에 학생 헤더가 등장하면 현재 학생 갱신 a = "" if pd.isna(row.iloc[0]) else str(row.iloc[0]).strip() m = student_header_re.match(a) if m: current_student = m.group(2).strip() # 이름만 사용 (학번 제외) continue if not current_student: continue # A~E열 = 월~금 시간표만 본다 for ci in range(min(5, len(row))): cell = row.iloc[ci] if pd.isna(cell): continue text = str(cell).strip() if not text or "학점" in text: continue # "(x시간)" 제거 → "과목명 n반" 형태만 남긴다 text_norm = remove_time_re.sub("", text).strip() if "반" not in text_norm: continue text_norm = re.sub(r"\s+", " ", text_norm) course_to_students[text_norm].add(current_student) student_to_courses[current_student].add(text_norm) # === 2) Build HTML page === template = """ 같은 수업 친구 찾기

같은 수업 친구 찾기

:)

과목(반)같은 수업 친구들 (이름만)
학생 이름을 입력하고 검색을 눌러주세요.
""" html = template.replace("__COURSE_TO_STUDENTS__", json.dumps({k: sorted(v) for k, v in course_to_students.items()}, ensure_ascii=False))\ .replace("__STUDENT_TO_COURSES__", json.dumps({k: sorted(v) for k, v in student_to_courses.items()}, ensure_ascii=False)) out_path = Path("/mnt/data/classmates_search_v2.html") out_path.write_text(html, encoding="utf-8") str(out_path)
